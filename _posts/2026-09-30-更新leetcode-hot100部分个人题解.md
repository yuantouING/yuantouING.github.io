## 题目
- 两数之和
- 字母异位词分组
- 最长连续序列

## 注意事项

两数之和
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int* two_sum(int* array, int nums_size, int target, int* return_size) {
	if (nums_size < 2) {
		*return_size = 0;
		return NULL;
	}
	for (int i = 0; i < nums_size - 1; i++) {
		for (int j = i + 1; j < nums_size; j++) {
			if (array[i] + array[j] == target) {
				int* result = calloc(2, sizeof(int));
				result[0] = i;
				result[1] = j;
				*return_size = 2;
				return result;
			}
		}
	}
	*return_size = 0;
	return NULL;
}
int main() {
	int* nums = calloc(1000, sizeof(int));
	char line[1000];
	fgets(line, sizeof(line), stdin);
	int nums_size = 0;
	char* token = strtok(line, " \n\r\t");
	while (token != NULL) {
		nums[nums_size++] = atoi(token);
		token = strtok(NULL, " \n\r\t");
	}
	int target;
	scanf("%d", &target);
	int return_size = 0;
	int* ans = two_sum(nums, nums_size, target, &return_size);
	if (ans != NULL) {
		printf("%d %d\n", ans[0], ans[1]);
		free(ans);
	}
	else {
		printf("无解\n");
	}
	free(nums);
	return 0;
}
```

解题思路：暴力解法，双重循环两两匹配，判断是否存在两个元素满足两数之和等于target，存在则记录并返回结果，不存在则设置返回结果的大小为0。
注：处理输入时不指定nums_size，本题解采用fgets读取一整行再按空格解析字符串，进而转换为整数存入nums数组。


字母异位词分组
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int cmp(const void* a, const void* b) {
	return *(char*)a - *(char*)b;
}
char* strSort(char* s) {
	char* sort = calloc(strlen(s) + 1, sizeof(char));
	strcpy(sort, s);
	qsort(sort, strlen(sort), sizeof(char), cmp);
	return sort;
}
char*** groupAnagrams(char** strs, int size, int* return_size, int** return_col_size) {
	char** keys = calloc(size, sizeof(char*));
	for (int i = 0; i < size; i++) {
		keys[i] = strSort(strs[i]);
	}
	int* visited = calloc(size, sizeof(int));
	char*** res = calloc(size, sizeof(char**));
	*return_size = 0;
	*return_col_size = calloc(size, sizeof(int));
	for (int i = 0; i < size; i++) {
		if (visited[i] == 1) {
			continue;
		}
		int groupsize = 0;
		char** group = calloc(size, sizeof(char*));
		for (int j = i; j < size; j++) {
			if (visited[j] == 0 && strcmp(keys[j], keys[i]) == 0) {
				group[groupsize] = strs[j];
				visited[j] = 1;
				groupsize++;
			}
		}
		res[*return_size] = group;
		(*return_col_size)[*return_size] = groupsize;
		(*return_size)++;
	}
	for (int i = 0; i < size; i++) free(keys[i]);
	free(keys);
	free(visited);
	return res;
}
int main() {
	char** strs = calloc(1000, sizeof(char*));
	int size = 0;
	char* str = calloc(1000, sizeof(char));
	while (scanf("%s", str) == 1) {
		strs[size] = calloc(strlen(str) + 1, sizeof(char));
		strcpy(strs[size], str);
		size++;
	}
	int return_size;
	int* return_col_size;
	char*** ans = groupAnagrams(strs, size, &return_size, &return_col_size);
	for (int i = 0; i < return_size; i++) {
		for (int j = 0; j < return_col_size[i]; j++) {
			printf("%s", ans[i][j]);
			if (j < return_col_size[i] - 1) {
				printf(" ");
			}
		}
		printf("\n");
	}
	for (int i = 0; i < return_size; i++) free(ans[i]);
	free(ans);
	free(return_col_size);

	for (int i = 0; i < size; i++) free(strs[i]);
	free(strs);
	free(str);
	return 0;
}
```

解题思路：借助函数strSort对字符串排序作为key，借助visited记录字符串是否被分组，外层for循环遍历字符串数组，先验证字符串是否被分组，已被分组则跳过，
未分组则利用内层for循环将其与其本身以及其之后、未被分组的字符串各自的key进行对比，相同则分在同一组。
注：处理输入时不指定str_size，每读取到一个字符串就将读取的字符串复制给创建的strs数组。


最长连续序列
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int cmp(const void* a, const void* b) {
	return *(int*)a - *(int*)b;
}
int longest_solve(int* nums, int nums_size) {
	if (nums == NULL || nums_size == 0) {
		return 0;
	}
	qsort(nums, nums_size, sizeof(int), cmp);
	int curlength = 1;
	int maxlength = 1;
	for (int i = 1; i < nums_size; i++) {
		if (nums[i] == nums[i - 1]) {
			continue;
		}
		else if (nums[i] == nums[i - 1] + 1) {
			curlength++;
		}
		else {
			maxlength = curlength > maxlength ? curlength : maxlength;
			curlength = 1;
		}
	}
	maxlength = curlength > maxlength ? curlength : maxlength;
	return maxlength;
}
int main() {
	char line[1000];
	int* nums = calloc(1000, sizeof(int));
	int nums_size = 0;
	fgets(line, sizeof(line), stdin);
	char* token = strtok(line, " \n\r\t");
	while (token != NULL) {
		nums[nums_size++] = atoi(token);
		token = strtok(NULL, " \n\r\t");
	}
	int ans = longest_solve(nums, nums_size);
	printf("%d", ans);
	free(nums);
	return 0;
}
```

解题思路：先对数组排序，如果某元素与其前一个元素相等则跳过，如果某元素与其前一个元素相差1则更新当前长度，
其余情况就更新最大长度并设置当前长度为1，由于最大长度可能在数组末尾才出现，因此还需再返回前再次更新最大长度。
注：处理输入时不指定nums_size，本题解采用fgets读取一整行再按空格解析字符串，进而转换为整数存入nums数组。
